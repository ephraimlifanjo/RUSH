cmake_minimum_required(VERSION 3.18)
project(rush_native_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(rush-native-core rush_native_core.cpp)
if(MSVC)
  target_compile_options(rush-native-core PRIVATE /O2 /EHsc)
else()
  target_compile_options(rush-native-core PRIVATE -O2)
endif()
